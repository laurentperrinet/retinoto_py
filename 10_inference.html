In this script, we evaluate the performance of different convolutional models on the validation dataset. We implement a simple inference pipeline: loading parameters, preparing data, loading a pre-trained model, and calculating accuracy.

This is a fundamental step to understand how a classical model behaves before introducing modifications such as masking or foveation.

# 7. Implementation of the evaluation loop
class_to_idx = val_dataset.class_to_idx

# Metrics tracking
correct_predictions = 0
total_predictions = 0

print(f"Starting evaluation on {len(val_loader.dataset)} images...")
print("-" * 50)

# Iterating over the validation loader
for images, true_dxs in fovea.tqdm(val_loader, desc="Evaluating"):
    images = images.to(args.device)
    true_dxs = true_dxs.to(args.device)

    # Disable gradient computation for inference to save memory and time
    with torch.no_grad():
        outputs = model(images)
        _, predicted_labels = torch.max(outputs, 1)

    # Extract labels and check correctness
    predicted_label_idx = predicted_labels[0].item()
    true_dx = true_dxs[0].item()
    predicted_class_name = val_dataset.idx_to_class[predicted_label_idx]
    true_class_name = val_dataset.idx_to_class[true_dx]
    is_correct = (predicted_label_idx == true_dx)
    if is_correct:
        correct_predictions += 1
    total_predictions += 1

    status = "✅ Correct" if is_correct else "❌ Incorrect"
    if torch.rand(1) < .0001: print(f"True: {true_class_name:<20} ({idx_to_label[true_dx]}) | Predicted: {predicted_class_name:<20} (({idx_to_label[predicted_label_idx]})) | {status}")

print("-" * 50)
accuracy = 100 * correct_predictions / total_predictions
print("Evaluation complete.")
print(f"Accuracy: {correct_predictions}/{total_predictions} ({accuracy:.2f}%)")
